cmake_minimum_required(VERSION 3.20)
project(MLCookbookGraphCapture LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This initial graph-capture helper supports Windows x64 only.")
endif()
add_executable(llama-graph-capture graph_capture.cpp)
target_compile_features(llama-graph-capture PRIVATE cxx_std_17)
target_include_directories(llama-graph-capture PRIVATE vendor)
target_link_libraries(llama-graph-capture PRIVATE bcrypt)
target_compile_definitions(llama-graph-capture PRIVATE UNICODE _UNICODE)
if(MSVC)
    target_compile_options(llama-graph-capture PRIVATE /W4 /EHsc /utf-8)
    set_property(TARGET llama-graph-capture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(llama-graph-capture PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
